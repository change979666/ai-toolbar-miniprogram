### 4.2 AI对话模块增强
**消息处理逻辑补充：**
```javascript
// 消息管理器核心逻辑
class MessageManager {
  constructor() {
    this.messages = uni.getStorageSync('chatHistory') || []
  }

  addMessage(message) {
    // 实现消息去重和敏感词过滤
    if (!this.isDuplicate(message) && !this.hasSensitiveWord(message)) {
      this.messages.push(message)
      this.persistMessages()
    }
  }

  persistMessages() {
    // 分页存储策略（每50条存一个文档）
    const chunks = chunk(this.messages, 50)
    chunks.forEach((chunk, index) => {
      wx.cloud.database().collection('chat_history').doc(`user_${uid}_part${index}`)
        .update({ data: { messages: chunk } })
    })
  }

  // 消息滚动优化
  scrollToBottom() {
    this.$nextTick(() => {
      const query = uni.createSelectorQuery().in(this)
      query.select('.message-container').boundingClientRect(res => {
        this.scrollTop = res.height + 100
      }).exec()
    })
  }
}
```

async function handleStreamResponse(response) {
  const reader = response.body.getReader()
  const decoder = new TextDecoder()
  let done = false
  
  while (!done) {
    const { value, done: streamDone } = await reader.read()
    done = streamDone
    const chunk = decoder.decode(value)
    this.partialResponse += chunk
    this.messages[this.messages.length - 1].content = this.partialResponse
    this.$forceUpdate()
  }
}
